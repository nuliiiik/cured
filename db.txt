-- 1. Создание базы данных
CREATE DATABASE CuredBase;
GO

USE CuredBase;
GO

-- 2. ТАБЛИЦА ПОЛЬЗОВАТЕЛЕЙ (dbo.Users) [cite: 2, 5]
CREATE TABLE dbo.Users (
    UserID INT IDENTITY(1,1) PRIMARY KEY,
    FullName NVARCHAR(255) NOT NULL, -- ФИО
    Login NVARCHAR(100) NOT NULL UNIQUE, -- Логин (уникальный)
    Password NVARCHAR(255) NOT NULL, -- Пароль
    Phone NVARCHAR(20) NOT NULL,     -- Номер телефона
    Role NVARCHAR(50) DEFAULT 'user' -- Роль (admin/user)
);

-- 3. ТАБЛИЦА МОТОЦИКЛОВ (dbo.Motorcycles) [cite: 2, 5]
CREATE TABLE dbo.Motorcycles (
    MotorcycleID INT IDENTITY(1,1) PRIMARY KEY,
    Brand NVARCHAR(100) NOT NULL,    -- Марка
    Model NVARCHAR(100) NOT NULL,    -- Модель
    Price DECIMAL(18, 2) NOT NULL,   -- Цена
    Year INT NOT NULL,               -- Год выпуска
    Description NVARCHAR(MAX),       -- Описание
    ImageURL NVARCHAR(MAX),          -- Путь к фото
    Quantity INT NOT NULL DEFAULT 0  -- Количество в наличии
);

-- 4. ТАБЛИЦА ЗАПЧАСТЕЙ (dbo.Parts) [cite: 2, 5]
CREATE TABLE dbo.Parts (
    PartID INT IDENTITY(1,1) PRIMARY KEY,
    PartName NVARCHAR(255) NOT NULL, -- Название запчасти
    Brand NVARCHAR(100) NOT NULL,    -- Совместимая марка
    Price DECIMAL(18, 2) NOT NULL,   -- Цена
    Description NVARCHAR(MAX),       -- Описание
    ImageURL NVARCHAR(MAX),          -- Путь к фото
    Quantity INT NOT NULL DEFAULT 0  -- Количество в наличии
);

-- 5. ТАБЛИЦА КОРЗИНЫ (dbo.Cart) [cite: 2, 5]
CREATE TABLE dbo.Cart (
    CartID INT IDENTITY(1,1) PRIMARY KEY,
    UserID INT NOT NULL,             -- Кто добавил
    ProductID INT NOT NULL,          -- Что добавил (ID мото или запчасти)
    ProductType NVARCHAR(10) NOT NULL, -- Тип: 'moto' или 'part'
    Count INT NOT NULL DEFAULT 1,    -- Сколько штук
    FOREIGN KEY (UserID) REFERENCES dbo.Users(UserID) ON DELETE CASCADE
);

-- 6. ТАБЛИЦА ЗАКАЗОВ (dbo.Orders) [cite: 2, 5]
CREATE TABLE dbo.Orders (
    OrderID INT IDENTITY(1,1) PRIMARY KEY,
    UserID INT NOT NULL,
    OrderDate DATETIME DEFAULT GETDATE(),
    TotalAmount DECIMAL(18, 2) NOT NULL,
    Status NVARCHAR(50) DEFAULT 'Ожидает',
    FOREIGN KEY (UserID) REFERENCES dbo.Users(UserID)
);

-- 7. ТАБЛИЦА СОСТАВА ЗАКАЗА (dbo.OrderItems) [cite: 2, 5]
CREATE TABLE dbo.OrderItems (
    OrderItemID INT IDENTITY(1,1) PRIMARY KEY,
    OrderID INT NOT NULL,
    ProductID INT NOT NULL,
    ProductType NVARCHAR(10) NOT NULL,
    Quantity INT NOT NULL,
    PriceAtOrder DECIMAL(18, 2) NOT NULL, -- Цена на момент покупки
    FOREIGN KEY (OrderID) REFERENCES dbo.Orders(OrderID) ON DELETE CASCADE
);
GO
