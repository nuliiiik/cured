-- 1. Создание базы данных
IF NOT EXISTS (SELECT * FROM sys.databases WHERE name = 'cursed')
BEGIN
    CREATE DATABASE [cursed];
END
GO

USE [cursed];
GO

-- 2. Таблица пользователей (соответствует логике в login.cs и profile.cs)
CREATE TABLE [dbo].[Users] (
    [UserID]   INT IDENTITY(1,1) PRIMARY KEY,
    [Login]    NVARCHAR(100) NOT NULL UNIQUE,
    [Password] NVARCHAR(255) NOT NULL,
    [FullName] NVARCHAR(255) NULL,
    [Phone]    NVARCHAR(20)  NULL,
    [Role]     NVARCHAR(50)  DEFAULT 'user' -- 'admin' или 'user'
);

-- 3. Таблица мотоциклов (соответствует вкладке tabMotocycles)
CREATE TABLE [dbo].[Motorcycles] (
    [MotorcycleID] INT IDENTITY(1,1) PRIMARY KEY,
    [Brand]        NVARCHAR(100) NOT NULL,
    [Model]        NVARCHAR(100) NOT NULL,
    [Year]         INT NOT NULL,
    [Price]        DECIMAL(18, 2) NOT NULL,
    [Quantity]     INT DEFAULT 0, -- Доступно для покупки
    [ImageURL]     NVARCHAR(MAX), -- Путь к картинке (напр. /images/motocycles/Honda.png)
    [Description]  NVARCHAR(MAX),
    [Stock]        INT DEFAULT 0  -- Общий остаток на складе
);

-- 4. Таблица запчастей (соответствует вкладке tabParts)
CREATE TABLE [dbo].[Parts] (
    [PartID]    INT IDENTITY(1,1) PRIMARY KEY,
    [PartName]  NVARCHAR(255) NOT NULL,
    [Brand]     NVARCHAR(100) NOT NULL, -- Совместимая марка
    [Price]     DECIMAL(18, 2) NOT NULL,
    [Quantity]  INT DEFAULT 0,
    [ForModels] NVARCHAR(MAX), -- Текстовое описание совместимых моделей
    [ImageURL]  NVARCHAR(MAX),
    [Stock]     INT DEFAULT 0
);

-- 5. Таблица корзины
CREATE TABLE [dbo].[Cart] (
    [CartID]       INT IDENTITY(1,1) PRIMARY KEY,
    [UserID]       INT NOT NULL,
    [MotorcycleID] INT NULL,
    [PartID]       INT NULL,
    [Quantity]     INT DEFAULT 1,
    [AddedDate]    DATETIME DEFAULT GETDATE(),
    FOREIGN KEY ([UserID]) REFERENCES [dbo].[Users]([UserID]) ON DELETE CASCADE,
    FOREIGN KEY ([MotorcycleID]) REFERENCES [dbo].[Motorcycles]([MotorcycleID]),
    FOREIGN KEY ([PartID]) REFERENCES [dbo].[Parts]([PartID])
);

-- 6. Таблица заказов (поля из скриншота 4163)
CREATE TABLE [dbo].[Orders] (
    [OrderID]        INT IDENTITY(1,1) PRIMARY KEY,
    [UserID]         INT NOT NULL,
    [OrderDate]      DATETIME DEFAULT GETDATE(),
    [TotalAmount]    DECIMAL(18, 2) NOT NULL,
    [Status]         NVARCHAR(50) DEFAULT N'Ожидает',
    [Address]        NVARCHAR(MAX),
    [Phone]          NVARCHAR(20),
    [PaymentMethod]  NVARCHAR(100),
    [DeliveryMethod] NVARCHAR(100),
    FOREIGN KEY ([UserID]) REFERENCES [dbo].[Users]([UserID])
);

-- 7. Детали заказа
CREATE TABLE [dbo].[OrderItems] (
    [OrderItemID]  INT IDENTITY(1,1) PRIMARY KEY,
    [OrderID]      INT NOT NULL,
    [MotorcycleID] INT NULL,
    [PartID]       INT NULL,
    [Quantity]     INT NOT NULL,
    [Price]        DECIMAL(18, 2) NOT NULL,
    FOREIGN KEY ([OrderID]) REFERENCES [dbo].[Orders]([OrderID]) ON DELETE CASCADE
);
GO
